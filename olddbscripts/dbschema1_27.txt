  `idCompany` INT NULL,
  `primaryColor` VARCHAR(45) NULL,
  `secondaryColor` VARCHAR(45) NULL,
  `tertiaryColor` VARCHAR(45) NULL,
  `brandLogo` VARCHAR(45) NULL,
